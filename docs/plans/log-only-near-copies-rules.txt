const IMPL = [CABLE, SMITH, MACH, BB, DB, KB, BAND]
const out: [string, string][] = [], skipped: string[] = [], unmapped: string[] = []
for (const r of rows) {
  if (r.section.startsWith('Cardio')) { skipped.push(r.name); continue }
  if (SKIP.test(r.name)) { skipped.push(r.name); continue }
  const implOf = IMPL.find(re => re.test(r.name)) ?? null
  const rule = R.find(([mv, im]) => mv.test(r.name) && (im ? im === implOf : !implOf || (implOf === MACH && /press|curl|extension|hyper|back ext/i.test(r.name) && /leg press|hack|belt|leg curl|extension|hyper/i.test(r.name)) ))
  if (!rule || !names.has(rule[2])) { unmapped.push(r.name); continue }
  out.push([r.name, rule[2]])
}
writeFileSync(process.argv[3], JSON.stringify(out))
const g: Record<string, string[]> = {}; for (const [n, p] of out) (g[p] ??= []).push(n)
console.log(`mapped ${out.length}, skipped ${skipped.length}, unmapped ${unmapped.length}`)
if (process.argv[4] === 'groups') for (const [p, ns] of Object.entries(g).sort()) console.log(`\n## ${p} (${ns.length}): ${ns.join(' | ')}`)
if (process.argv[4] === 'unmapped') console.log(unmapped.join(' | '))
